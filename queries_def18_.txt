CREATE TABLE def18_Company_master (
    Com_id INT PRIMARY KEY,
    com_name VARCHAR(100) NOT NULL
);

CREATE TABLE def18_Item_Master (
    Item_id INT PRIMARY KEY,
    Item_name VARCHAR(100) NOT NULL,
    price DECIMAL(10, 2) NOT NULL,
    company_code INT,
    FOREIGN KEY (company_code) REFERENCES def18_Company_master(Com_id)
);

SELECT 
    i.Item_id,
    i.Item_name,
    i.price,
    i.company_code,
    c.com_name AS producer_company
FROM def18_Item_Master i
JOIN def18_Company_master c 
  ON i.company_code = c.Com_id;
  
  
SELECT 
    i.Item_name,
    i.price,
    c.com_name
FROM def18_Item_Master i
JOIN def18_Company_master c 
  ON i.company_code = c.Com_id;
  
  
SELECT 
    c.com_name,
    AVG(i.price) AS average_price
FROM def18_Company_master c
JOIN def18_Item_Master i 
  ON c.Com_id = i.company_code
GROUP BY c.Com_id, c.com_name;


SELECT 
    c.com_name,
    AVG(i.price) AS average_price
FROM def18_Company_master c
JOIN def18_Item_Master i 
  ON c.Com_id = i.company_code
GROUP BY c.Com_id, c.com_name
HAVING AVG(i.price) >= 350;
