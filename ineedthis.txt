SELECT DISTINCT c.cname
FROM def10_customer c
LEFT JOIN def_10deposit d ON c.cname = d.cname
LEFT JOIN def10_borrow br ON c.cname = br.cname
LEFT JOIN def10_branch bd ON d.bname = bd.bname
LEFT JOIN def10_branch bb ON br.bname = bb.bname
WHERE c.city = 'Mumbai' 
  AND (bd.city = 'Nagpur' OR bb.city = 'Nagpur');
